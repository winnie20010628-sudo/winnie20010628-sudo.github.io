Algorithm 1: calculate_iqr

Time complexity: O(n)

Justification:
Let n = length of sorted_data.
- Finding mid and calling median() only uses indexing → O(1)
- Slicing sorted_data[:mid] and the upper half each copy
  about n/2 elements → O(n)
- No nested loops over the data
Therefore overall time complexity is O(n).


Algorithm 2: remove_duplicates

Time complexity: O(n²)

Justification:
Let n = length of arr.
- The for-loop runs n times
- Inside the loop, "if num not in unique_list" checks up to
  n items, which is O(n)
- So total work = O(n) × O(n) = O(n²)


Algorithm 3: cal_ops

Time complexity: O(log N)

Justification:
Each loop sets i = i // 2, so i is halved every time.
Starting from N, values go N, N/2, N/4, ... until 1.
Number of iterations ≈ log₂(N).
Each iteration does constant work.
Therefore time complexity is O(log N).

def sum_of_digits(n):
    if n < 10:                 # base case
        return n
    else:                      # recursive case
        return (n % 10) + sum_of_digits(n // 10)


# Test
print(sum_of_digits(12345))

Working:
sum_of_digits(12345)
= 5 + sum_of_digits(1234)
= 5 + 4 + sum_of_digits(123)
= 5 + 4 + 3 + sum_of_digits(12)
= 5 + 4 + 3 + 2 + sum_of_digits(1)
= 5 + 4 + 3 + 2 + 1
= 15

Output: 15

class Stack:
    def __init__(self):
        self.items = []
    def is_empty(self):
        return self.items == []
    def push(self, item):
        self.items.append(item)
    def pop(self):
        return self.items.pop()

class Queue:
    def __init__(self):
        self.items = []
    def is_empty(self):
        return self.items == []
    def enqueue(self, item):
        self.items.append(item)
    def dequeue(self):
        return self.items.pop(0)


student_id = "21405678"
stack = Stack()
queue = Queue()

# Step 1: push each digit onto the stack
for d in student_id:
    stack.push(int(d))

# Step 2: empty stack into queue
while not stack.is_empty():
    queue.enqueue(stack.pop())

# Step 3: empty the queue
while not queue.is_empty():
    print(queue.dequeue(), end=" ")

class Node:
    def __init__(self, data):
        self.data = data
        self.next = None

    def get_data(self):
        return self.data

    def get_next(self):
        return self.next

    def set_data(self, data):
        self.data = data

    def set_next(self, next_node):
        self.next = next_node


class UnorderedList:
    def __init__(self):
        self.head = None

    def is_empty(self):
        return self.head == None

    def add(self, item):
        temp = Node(item)
        temp.set_next(self.head)
        self.head = temp

    def size(self):
        current = self.head
        count = 0
        while current != None:
            count = count + 1
            current = current.get_next()
        return count

    def print_list(self):
        current = self.head
        while current != None:
            print(current.get_data(), end=" ")
            current = current.get_next()
        print()

    def search(self, item):
        current = self.head
        while current != None:
            if current.get_data() == item:
                return True
            current = current.get_next()
        return False

----- Bubble Sort -----

def bubble_sort(arr):
    n = len(arr)
    for pass_num in range(1, n):
        for i in range(n - pass_num):
            if arr[i] > arr[i + 1]:
                arr[i], arr[i + 1] = arr[i + 1], arr[i]
        print("After pass", pass_num, ":", arr)

data = [2, 1, 4, 0, 5, 6, 7, 8]
print("Initial:", data)
bubble_sort(data)

----- Quick Sort -----

def quick_sort(arr):
    if len(arr) <= 1:
        return arr
    pivot = arr[-1]
    left = [x for x in arr[:-1] if x <= pivot]
    right = [x for x in arr[:-1] if x > pivot]
    return quick_sort(left) + [pivot] + quick_sort(right)
